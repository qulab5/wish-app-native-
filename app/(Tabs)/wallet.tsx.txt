import React, { useState } from 'react';
import {
  View, Text, StyleSheet, ScrollView, TouchableOpacity,
  Alert, Clipboard,
} from 'react-native';
import { useAuth } from '../../context/AuthContext';
import { COLORS as C, API_BASE } from '../../constants';

const MOCK_ACTIVITY = [
  { id: '1', label: 'Mining Reward', amount: '+8,000', pts: true, date: 'Today', icon: '⛏', positive: true },
  { id: '2', label: 'Spin Win', amount: '+2,500', pts: true, date: 'Today', icon: '🎡', positive: true },
  { id: '3', label: 'Twitter Task', amount: '+2 WISH', pts: false, date: 'Yesterday', icon: '𝕏', positive: true },
  { id: '4', label: 'Convert', amount: '-1,000 pts → +1 WISH', pts: false, date: 'Yesterday', icon: '🔄', positive: false },
  { id: '5', label: 'Referral Bonus', amount: '+2,000', pts: true, date: '2 days ago', icon: '👥', positive: true },
];

export default function WalletScreen() {
  const { user, updateUser } = useAuth();
  const [tab, setTab] = useState<'assets' | 'activity'>('assets');
  const [refreshing, setRefreshing] = useState(false);

  const pts = user?.pts ?? 0;
  const tokens = user?.tokens ?? 0;
  const walletAddr = user?.walletAddress || 'ukpKw2...8PTWrF';

  const copyAddress = () => {
    Clipboard.setString(walletAddr);
    Alert.alert('Copied!', 'Wallet address copied to clipboard.');
  };

  const refresh = async () => {
    if (!user || refreshing) return;
    setRefreshing(true);
    try {
      const r = await fetch(`${API_BASE}/api/user?email=${encodeURIComponent(user.email)}`);
      const d = await r.json();
      if (d.user) updateUser(d.user);
    } catch {}
    setRefreshing(false);
  };

  return (
    <View style={styles.container}>
      <View style={styles.header}>
        <View style={{ width: 36 }} />
        <Text style={styles.headerTitle}>Wish Network</Text>
        <View style={{ width: 36 }} />
      </View>

      <ScrollView showsVerticalScrollIndicator={false} contentContainerStyle={{ padding: 16, paddingBottom: 40 }}>

        {/* Wallet Title Row */}
        <View style={{ flexDirection: 'row', alignItems: 'center', marginBottom: 14 }}>
          <View style={styles.walletIcon}>
            <Text style={{ fontSize: 28 }}>🦅</Text>
          </View>
          <View style={{ flex: 1 }}>
            <Text style={styles.balLabel}>YOUR BALANCE</Text>
            <Text style={styles.walletTitle}>Wish Wallet</Text>
          </View>
          <TouchableOpacity style={styles.refreshBtn} onPress={refresh}>
            <Text style={{ fontSize: 16 }}>🔄</Text>
          </TouchableOpacity>
        </View>

        {/* Balance Card */}
        <View style={styles.balanceCard}>
          <View style={{ flexDirection: 'row', alignItems: 'center', marginBottom: 8 }}>
            <View style={styles.coinLogo}>
              <Text style={{ fontSize: 32 }}>🦅</Text>
            </View>
            <View style={{ marginLeft: 14 }}>
              <Text style={styles.coinLabel}>WISH COINS</Text>
              <View style={{ flexDirection: 'row', alignItems: 'baseline', gap: 8 }}>
                <Text style={styles.coinValue}>{tokens.toLocaleString()}</Text>
                <Text style={{ color: C.gold, fontWeight: '700', fontSize: 16 }}>WISH</Text>
              </View>
            </View>
          </View>

          {/* Address */}
          <View style={styles.addrRow}>
            <Text style={{ fontSize: 12, color: C.muted }}>◇</Text>
            <Text style={styles.addrText} numberOfLines={1}>{walletAddr}</Text>
            <TouchableOpacity style={styles.copyBtn} onPress={copyAddress}>
              <Text style={{ color: C.brand, fontWeight: '700', fontSize: 13 }}>Copy</Text>
            </TouchableOpacity>
          </View>

          {/* Send / Receive */}
          <View style={styles.actionRow}>
            <TouchableOpacity
              style={styles.actionBtn}
              onPress={() => Alert.alert('Send', 'Token sending coming soon!')}
            >
              <Text style={styles.actionBtnText}>↑ Send</Text>
            </TouchableOpacity>
            <TouchableOpacity
              style={[styles.actionBtn, styles.actionBtnGhost]}
              onPress={() => Alert.alert('Receive', `Your wallet address:\n\n${walletAddr}`)}
            >
              <Text style={[styles.actionBtnText, { color: C.white }]}>↓ Receive</Text>
            </TouchableOpacity>
          </View>
        </View>

        {/* Assets / Activity Tabs */}
        <View style={styles.tabRow}>
          <TouchableOpacity
            style={[styles.tabBtn, tab === 'assets' && styles.tabBtnActive]}
            onPress={() => setTab('assets')}
          >
            <Text style={[styles.tabTxt, tab === 'assets' && styles.tabTxtActive]}>Assets</Text>
          </TouchableOpacity>
          <TouchableOpacity
            style={[styles.tabBtn, tab === 'activity' && styles.tabBtnActive]}
            onPress={() => setTab('activity')}
          >
            <Text style={[styles.tabTxt, tab === 'activity' && styles.tabTxtActive]}>Activity</Text>
          </TouchableOpacity>
        </View>

        {tab === 'assets' ? (
          <>
            <View style={styles.assetRow}>
              <View style={styles.assetIcon}>
                <Text style={{ fontSize: 28 }}>🦅</Text>
              </View>
              <View style={{ flex: 1, marginLeft: 12 }}>
                <Text style={{ color: C.white, fontWeight: '700', fontSize: 15 }}>Wish Coin</Text>
                <Text style={styles.assetSub}>WISH</Text>
              </View>
              <View style={{ alignItems: 'flex-end' }}>
                <Text style={{ color: C.gold, fontWeight: '800', fontSize: 18 }}>{tokens.toLocaleString()}</Text>
                <Text style={styles.changeText}>▲ 0.00%</Text>
              </View>
            </View>

            <View style={styles.assetRow}>
              <View style={[styles.assetIcon, { backgroundColor: 'rgba(245,158,11,0.15)', borderColor: 'rgba(245,158,11,0.3)' }]}>
                <Text style={{ fontSize: 28 }}>⭐</Text>
              </View>
              <View style={{ flex: 1, marginLeft: 12 }}>
                <Text style={{ color: C.white, fontWeight: '700', fontSize: 15 }}>Wish Points</Text>
                <Text style={styles.assetSub}>PTS</Text>
              </View>
              <View style={{ alignItems: 'flex-end' }}>
                <Text style={{ color: C.gold, fontWeight: '800', fontSize: 18 }}>{pts.toLocaleString()}</Text>
                <Text style={styles.changeText}>≈ {Math.floor(pts / 1000)} WISH</Text>
              </View>
            </View>
          </>
        ) : (
          <>
            {MOCK_ACTIVITY.map(a => (
              <View key={a.id} style={styles.actRow}>
                <View style={styles.actIcon}>
                  <Text style={{ fontSize: 20 }}>{a.icon}</Text>
                </View>
                <View style={{ flex: 1, marginLeft: 12 }}>
                  <Text style={{ color: C.white, fontWeight: '600', fontSize: 14 }}>{a.label}</Text>
                  <Text style={styles.assetSub}>{a.date}</Text>
                </View>
                <Text style={{ color: a.positive ? C.green : C.red, fontWeight: '700', fontSize: 14 }}>
                  {a.amount} {a.pts ? 'pts' : ''}
                </Text>
              </View>
            ))}
          </>
        )}

      </ScrollView>
    </View>
  );
}
