import React, { useState, useEffect, useRef } from 'react';
import {
  View, Text, StyleSheet, ScrollView, TouchableOpacity,
  Animated, Alert, Modal,
} from 'react-native';
import { router } from 'expo-router';
import { useAuth } from '../../context/AuthContext';
import { COLORS as C, API_BASE, MINE_SESSION_S, MINE_PTS_HOUR, MINE_BOOST_PTS, MINE_MAX_BOOST } from '../../constants';

export default function HomeScreen() {
  const { user, updateUser, logout } = useAuth();
  const [menuOpen, setMenuOpen] = useState(false);
  const [mineActive, setMineActive] = useState(false);
  const [mineStart, setMineStart] = useState<number | null>(user?.mineStart || null);
  const [earnedSoFar, setEarnedSoFar] = useState(0);
  const [boostsLeft, setBoostsLeft] = useState(user?.boostsLeft ?? MINE_MAX_BOOST);
  const [mineComplete, setMineComplete] = useState(false);
  const [timeLeft, setTimeLeft] = useState('');
  const pulseAnim = useRef(new Animated.Value(1)).current;

  const pts = user?.pts ?? 0;
  const tokens = user?.tokens ?? 0;
  const spinsLeft = user?.spinsLeft ?? 1;

  useEffect(() => {
    if (!mineActive) return;
    const loop = Animated.loop(
      Animated.sequence([
        Animated.timing(pulseAnim, { toValue: 1.06, duration: 800, useNativeDriver: true }),
        Animated.timing(pulseAnim, { toValue: 1, duration: 800, useNativeDriver: true }),
      ])
    );
    loop.start();
    return () => loop.stop();
  }, [mineActive]);

  useEffect(() => {
    if (!mineStart) return;
    const iv = setInterval(() => {
      const elapsed = (Date.now() / 1000) - mineStart;
      const progress = Math.min(elapsed / MINE_SESSION_S, 1);
      const earned = Math.floor(progress * MINE_PTS_HOUR * 4);
      setEarnedSoFar(earned);
      if (elapsed >= MINE_SESSION_S) {
        setMineActive(false);
        setMineComplete(true);
        setTimeLeft('Complete!');
        clearInterval(iv);
      } else {
        setMineActive(true);
        setMineComplete(false);
        const remaining = MINE_SESSION_S - elapsed;
        const h = Math.floor(remaining / 3600);
        const m = Math.floor((remaining % 3600) / 60);
        const s = Math.floor(remaining % 60);
        setTimeLeft(`${h}h ${m}m ${s}s`);
      }
    }, 1000);
    return () => clearInterval(iv);
  }, [mineStart]);

  const startMining = () => {
    const now = Date.now() / 1000;
    setMineStart(now);
    setMineActive(true);
    setMineComplete(false);
    setBoostsLeft(MINE_MAX_BOOST);
    setEarnedSoFar(0);
    updateUser({ mineStart: now, boostsLeft: MINE_MAX_BOOST });
    if (user) {
      fetch(`${API_BASE}/api/user`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ id: user.id, email: user.email, mineStart: now }),
      }).catch(() => {});
    }
  };

  const claimMining = async () => {
    if (!user) return;
    const newPts = pts + earnedSoFar;
    updateUser({ pts: newPts, mineStart: undefined });
    setMineStart(null);
    setMineActive(false);
    setMineComplete(false);
    setEarnedSoFar(0);
    try {
      await fetch(`${API_BASE}/api/user`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ id: user.id, email: user.email, pts: newPts, mineStart: null }),
      });
    } catch {}
    Alert.alert('🎉 Claimed!', `+${earnedSoFar.toLocaleString()} pts added to your balance!`);
  };

  const boostMining = () => {
    if (boostsLeft <= 0) { Alert.alert('No boosts left', 'Max 5 boosts per session.'); return; }
    Alert.alert('Watch Ad', 'Watch an ad to boost your mining?', [
      { text: 'Cancel', style: 'cancel' },
      {
        text: 'Watch', onPress: () => {
          const newBoosts = boostsLeft - 1;
          setBoostsLeft(newBoosts);
          setEarnedSoFar(e => e + MINE_BOOST_PTS);
          updateUser({ boostsLeft: newBoosts });
        }
      },
    ]);
  };

  const menuItems = [
    { icon: '👤', label: 'Profile' },
    { icon: '🏆', label: 'Leader Board' },
    { icon: '👥', label: 'Referral Team' },
    { icon: '🔔', label: 'Announcement' },
    { icon: '📄', label: 'White Paper' },
    { icon: '❓', label: 'Support Portal', badge: 'SOON' },
    { icon: '🛡', label: 'Security' },
  ];

  return (
    <View style={styles.container}>
      {/* Header */}
      <View style={styles.header}>
        <TouchableOpacity onPress={() => setMenuOpen(true)} style={{ padding: 4 }}>
          <View style={{ gap: 5 }}>
            {[0, 1, 2].map(i => <View key={i} style={styles.menuLine} />)}
          </View>
        </TouchableOpacity>
        <Text style={styles.headerTitle}>Wish Network</Text>
        <View style={{ width: 36 }} />
      </View>

      <ScrollView showsVerticalScrollIndicator={false}>
        {/* Profile Card */}
        <View style={styles.profileCard}>
          <View style={styles.avatar}>
            <Text style={{ fontSize: 28, fontWeight: '800', color: C.white }}>
              {(user?.name || 'U')[0].toUpperCase()}
            </Text>
          </View>
          <View style={{ flex: 1 }}>
            <Text style={styles.profileName}>{user?.name}</Text>
            <Text style={styles.profileSub}>🌐 · {user?.joined || 'Mar 2026'}</Text>
          </View>
          <View style={styles.verifiedBadge}>
            <Text style={{ fontSize: 11, color: C.green, fontWeight: '700' }}>✓✓ Verified</Text>
          </View>
        </View>

        {/* Stats Row */}
        <View style={styles.statsRow}>
          <StatCard icon="⭐" value={pts.toLocaleString()} label="Total Points" color={C.gold} />
          <StatCard icon="🦅" value={tokens.toLocaleString()} label="Tokens" color={C.brand} />
          <StatCard icon="🎡" value={String(spinsLeft)} label="Spins Left" color={C.cyan} />
        </View>

        {/* Mining Card */}
        <View style={styles.card}>
          <View style={{ flexDirection: 'row', alignItems: 'center', marginBottom: 12 }}>
            <Text style={{ fontSize: 28, marginRight: 10 }}>⛏</Text>
            <View style={{ flex: 1 }}>
              <Text style={styles.cardTitle}>Start Mining</Text>
              <Text style={styles.cardSub}>2,000 pts/hr · 4 hr session</Text>
            </View>
          </View>

          <View style={{ flexDirection: 'row', gap: 12, marginBottom: 12 }}>
            <View style={[styles.miniCard, { flex: 1 }]}>
              <Text style={styles.miniLabel}>Earned So Far</Text>
              <Text style={[styles.miniValue, { color: C.brand }]}>{earnedSoFar.toLocaleString()} pts</Text>
            </View>
            <View style={[styles.miniCard, { flex: 1 }]}>
              <Text style={styles.miniLabel}>Boosts Left</Text>
              <Text style={[styles.miniValue, { color: C.gold }]}>{boostsLeft}/{MINE_MAX_BOOST}</Text>
              <Text style={{ fontSize: 10, color: C.muted }}>this session</Text>
            </View>
          </View>

          {mineActive && (
            <View style={[styles.statusBox, { backgroundColor: 'rgba(16,185,129,0.1)', borderColor: 'rgba(16,185,129,0.3)' }]}>
              <Text style={{ color: C.green, fontWeight: '700' }}>⛏ Mining Active — {timeLeft} remaining</Text>
            </View>
          )}

          {mineComplete && (
            <>
              <View style={[styles.statusBox, { backgroundColor: 'rgba(16,185,129,0.1)', borderColor: 'rgba(16,185,129,0.3)', alignItems: 'center' }]}>
                <Text style={{ color: C.green, fontWeight: '700' }}>Mining Complete!</Text>
                <Text style={{ color: C.muted, fontSize: 12 }}>+{earnedSoFar.toLocaleString()} pts ready to claim</Text>
              </View>
              <TouchableOpacity style={[styles.bigBtn, { backgroundColor: C.green, marginTop: 10 }]} onPress={claimMining}>
                <Text style={styles.bigBtnText}>Claim {earnedSoFar.toLocaleString()} pts</Text>
              </TouchableOpacity>
            </>
          )}

          {!mineActive && !mineComplete && (
            <TouchableOpacity style={styles.bigBtn} onPress={startMining}>
              <Text style={styles.bigBtnText}>⛏ Start Mining</Text>
            </TouchableOpacity>
          )}

          {(mineActive || mineComplete) && (
            <TouchableOpacity
              style={[styles.bigBtn, { backgroundColor: 'rgba(245,158,11,0.15)', borderWidth: 1, borderColor: 'rgba(245,158,11,0.3)', marginTop: 8 }]}
              onPress={boostMining}
            >
              <Text style={[styles.bigBtnText, { color: C.gold }]}>⚡ Boost Mining {boostsLeft}/{MINE_MAX_BOOST}</Text>
            </TouchableOpacity>
          )}
        </View>

        <View style={{ height: 30 }} />
      </ScrollView>

      {/* Slide-out Menu */}
      <Modal visible={menuOpen} transparent animationType="fade">
        <TouchableOpacity style={styles.overlay} onPress={() => setMenuOpen(false)} activeOpacity={1}>
          <TouchableOpacity activeOpacity={1} style={styles.drawer}>
            <View style={styles.drawerHeader}>
              <Text style={[styles.headerTitle, { color: C.brand }]}>Wish Network</Text>
              <TouchableOpacity onPress={() => setMenuOpen(false)}>
                <Text style={{ color: C.muted, fontSize: 18 }}>✕</Text>
              </TouchableOpacity>
            </View>

            <View style={styles.drawerUser}>
              <View style={[styles.avatar, { width: 48, height: 48, borderRadius: 24 }]}>
                <Text style={{ fontSize: 20, fontWeight: '800', color: C.white }}>
                  {(user?.name || 'U')[0].toUpperCase()}
                </Text>
              </View>
              <View style={{ marginLeft: 12 }}>
                <Text style={{ color: C.white, fontWeight: '700', fontSize: 15 }}>{user?.name}</Text>
                <Text style={{ color: C.muted, fontSize: 12 }}>{user?.email}</Text>
              </View>
            </View>

            <View style={styles.divider} />

            {menuItems.map((item, i) => (
              <TouchableOpacity key={i} style={styles.menuItem} onPress={() => setMenuOpen(false)}>
                <Text style={{ fontSize: 18, width: 28 }}>{item.icon}</Text>
                <Text style={{ color: C.white, flex: 1, fontSize: 14, fontWeight: '500' }}>{item.label}</Text>
                {item.badge
                  ? <View style={styles.soonBadge}><Text style={{ fontSize: 10, color: C.brand, fontWeight: '700' }}>{item.badge}</Text></View>
                  : <Text style={{ color: C.muted }}>›</Text>
                }
              </TouchableOpacity>
            ))}

            <View style={{ flex: 1 }} />

            <TouchableOpacity
              style={styles.logoutBtn}
              onPress={() => {
                setMenuOpen(false);
                Alert.alert('Logout', 'Are you sure you want to logout?', [
                  { text: 'Cancel', style: 'cancel' },
                  { text: 'Yes, Logout', style: 'destructive', onPress: () => { logout(); router.replace('/'); } },
                ]);
              }}
            >
              <Text style={{ fontSize: 16, color: C.gold, fontWeight: '700' }}>👋 Logout</Text>
            </TouchableOpacity>
          </TouchableOpacity>
        </TouchableOpacity>
      </Modal>
    </View>
  );
}

      <Text style={[styles.statValue, { color }]}>{value}</Text>
      <Text style={styles.statLabel}>{label}</Text>
    </View>
  );
}
